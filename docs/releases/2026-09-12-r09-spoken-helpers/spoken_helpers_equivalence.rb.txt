strings = []
def literals(node, strings)
  return unless node.is_a?(Parser::AST::Node)
  strings << node.children.first if node.type == :str && node.children.first.match?(/[[:alpha:]]/) && node.children.first.include?(' ')
  node.children.each { |child| literals(child, strings) }
end

def outcome(klass, text, signal)
  instance = klass.new(text, answered_signal: signal)
  [instance.evidence, instance.observations]
rescue StandardError => error
  [:raised, error.class.name, error.message]
end
comparisons = 0
strings.product([nil, 'budget', 'decision_authority', 'urgency', 'lead_volume', 'contact_details', 'business_type', 'problem']).each do |text, signal|
  original = outcome(AiLeadEmployee::BaselineQualificationEvidenceExtractor, text, signal)
  current = outcome(AiLeadEmployee::QualificationEvidenceExtractor, text, signal)
  raise "Interpretation changed: #{[text, signal, original, current].inspect}" unless original == current
  comparisons += 1
end
